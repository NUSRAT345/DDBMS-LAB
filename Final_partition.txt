Q8 — Join Query
Write a Hive query to display student_name, course_name, marks for all students in the Computer Science department.

1. USE education_db;

CREATE TABLE IF NOT EXISTS courses (
    course_id     STRING,
    course_name   STRING,
    credits       INT,
    department    STRING
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/TeacherDetail/courses';

2. docker cp "C:\Users\User\Desktop\DDBMS Lab\course.csv" MyHadoop:/root/

3. LOAD DATA LOCAL INPATH '/root/course.csv' INTO TABLE courses;

4. SELECT 
    s.student_name,
    c.course_name,
    sr.marks
FROM student_results sr JOIN students s 
    ON sr.student_id = s.student_id
JOIN courses c 
    ON sr.subject_code = CAST(c.course_id AS STRING)
WHERE s.department = 'Computer Science';

Q9 — Aggregation (Average Marks per Department)

1. SELECT 
    s.department,
    AVG(sr.marks) AS avg_marks
FROM student_results sr
JOIN students s 
ON sr.student_id = s.student_id
WHERE sr.exam_year = 2024
GROUP BY s.department;



Q10 — Top Scorer in Spring 2024

1. WITH max_marks_cte AS (
    SELECT MAX(marks) AS max_marks
    FROM student_results
    WHERE exam_year = 2024 AND exam_session = 'Spring'
)
SELECT 
    sr.student_id,
    s.student_name,
    sr.subject_code,
    sr.marks
FROM student_results sr JOIN max_marks_cte mm 
    ON sr.marks = mm.max_marks
JOIN students s 
    ON sr.student_id = s.student_id
WHERE sr.exam_year = 2024 AND sr.exam_session = 'Spring';

10. CREATE DATABASE IF NOT EXISTS TeacherInfo;

USE TeacherInfo;

CREATE EXTERNAL TABLE IF NOT EXISTS Department (
    Dept_id INT,
    Dept_Name STRING,
    Building STRING
)
COMMENT 'Dept_Details'
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','  -- Changed from semicolon to comma (or your preferred delimiter)
STORED AS TEXTFILE
LOCATION '/TeacherDetail/department';

11.CREATE EXTERNAL TABLE IF NOT EXISTS Teacher (
    Teacher_id INT,
    Teacher_Name STRING,
    Email STRING,
    Dept_id INT
)
COMMENT 'Teacher_Details'
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','  -- Fixed typo in "TERMINATED" and added quotes around delimiter
STORED AS TEXTFILE
LOCATION '/TeacherDetail/teacher';

12. CREATE EXTERNAL TABLE IF NOT EXISTS Courses (
    course_id INT,
    course_name STRING,
    credit INT,
    dept_id INT,
    teacher_id INT
)
COMMENT 'Course information'
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/TeacherDetail/course';

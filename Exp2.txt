    exam_year INT,

STORED AS TEXTFILE;

Q7 — Create Additional Tables (Students & Courses)
-- Create students table
CREATE TABLE IF NOT EXISTS students (
    student_id INT,
    student_name STRING,
    dob STRING,
    department STRING
) STORED AS PARQUET;

-- Create courses table
CREATE TABLE IF NOT EXISTS courses (
    course_id INT,
    course_name STRING,
    credits INT,
    department STRING
) STORED AS PARQUET;

Q8 — Join Query

SELECT
    s.student_name,
    c.course_name,
    sr.marks
FROM student_results sr
JOIN students s ON sr.student_id = s.student_id
JOIN courses c ON sr.subject_code = CAST(c.course_id AS STRING)
WHERE s.department = 'Computer Science';

Q9 — Aggregation Query

SELECT
    s.department,
    AVG(sr.marks) AS avg_marks
FROM student_results sr
JOIN students s ON sr.student_id = s.student_id
WHERE sr.exam_year = 2024
GROUP BY s.department;

Q10 — Top Scorer Query (Bonus)

WITH max_marks_cte AS (
    SELECT MAX(marks) AS max_marks
    FROM student_results
    WHERE exam_year = 2024 AND exam_session = 'Spring'
)
SELECT
    sr.student_id,
    s.student_name,
    sr.subject_code,
    sr.marks
FROM student_results sr
JOIN max_marks_cte mm ON sr.marks = mm.max_marks
JOIN students s ON sr.student_id = s.student_id
WHERE sr.exam_year = 2024 AND sr.exam_session = 'Spring';
